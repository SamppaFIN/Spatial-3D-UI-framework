Code Example

import { Button3D } from './src/controls/Button3D.js';

const button = new Button3D(scene, camera, [0, 0, 0], {
    label: 'Click Me',
    color: 0x4ecdc4,
    mode: 0,
    loading: false,
    pulseAnimation: false,
    rippleEffect: true,
    width: 3.0,
    height: 1.0,
    renderer: renderer,
    onClick: () => console.log('Clicked!')
});

function animate() {
    requestAnimationFrame(animate);
    button.update();
    scene3D.render();
}

API Reference

Property Type Default Description
label string 'Button' Button text
color number 0x4ecdc4 Button color (hex)
mode number 0 0=box, 1=sphere, 2=sacred
loading boolean false Show loading spinner
pulseAnimation boolean false Enable pulse glow
rippleEffect boolean true Enable click ripple
width number 2.0 Button width
height number 0.8 Button height
