Slider3D

Responsive 3D value slider with gradient tracks and precise control

Live Demo

Value: 50

⚙️ Controls

Code Example

import { Slider3D } from './src/controls/Slider3D.js';

const slider = new Slider3D(scene, camera, [0, 0, 0], {
    min: 0,
    max: 100,
    value: 50,
    step: 1,
    width: 4.0,
    fillColor: 0x3366ff,
    trackColor: 0x444444,
    showValue: true,
    onChange: (slider, value) => {
        console.log('Value:', value);
    }
});

function animate() {
    requestAnimationFrame(animate);
    slider.update();
    scene3D.render();
}
