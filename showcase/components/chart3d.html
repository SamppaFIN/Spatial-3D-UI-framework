Chart3D

Data visualization panels with 3D presence and spatial integration

Live Demo

⚙️ Controls

Code Example

import { Chart3D } from './src/controls/Chart3D.js';

const chart = new Chart3D(scene, camera, [0, 0, 0], {
    chartType: 'line', // 'bar', 'pie', 'doughnut'
    width: 4.0,
    height: 3.0,
    data: {
        labels: ['Jan', 'Feb', 'Mar', 'Apr', 'May'],
        datasets: [{
            label: 'Sales',
            data: [65, 59, 80, 81, 56],
            backgroundColor: ['#6bb6ff', '#ff6b6b', '#4ecdc4', '#ffe66d', '#a8e6cf'],
            borderColor: '#6bb6ff'
        }]
    }
});

function animate() {
    requestAnimationFrame(animate);
    chart.update();
    scene3D.render();
}
