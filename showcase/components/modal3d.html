Immersive 3D dialogs with spatial presence and content integration

Live Demo

Code Example

import { Modal3D } from './src/controls/Modal3D.js';

const modal = new Modal3D(scene, camera, [0, 0, 0], {
    title: 'System Alert',
    width: 6.0,
    height: 4.0,
    htmlContent: '
Alert
This is a 3D modal.
',
    mode: 0,
    onClose: () => console.log('Modal closed')
});

// Trigger open
modal.open();

function animate() {
    requestAnimationFrame(animate);
    modal.update();
    scene3D.render();
}
