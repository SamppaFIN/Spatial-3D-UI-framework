TextDisplay3D

3D panel for displaying rich markdown content with spatial presence

Live Demo

⚙️ Controls

Code Example

import { TextDisplay3D } from './src/controls/TextDisplay3D.js';

const textPanel = new TextDisplay3D(scene, camera, [0, 0, 0], {
    content: '# Hello\nThis is **spatial text**.',
    width: 4.0,
    height: 3.0,
    style: 'post-it',
    mode: 0
});

function animate() {
    requestAnimationFrame(animate);
    textPanel.update();
    scene3D.render();
}
