Accordion3D

Expandable 3D content panels for organized information display

Live Demo

⚙️ Controls

Code Example

import { Accordion3D } from './src/controls/Accordion3D.js?v=2';

const accordion = new Accordion3D(scene, camera, [0, 2, 0], {
    width: 5.0,
    items: [
        {
            title: 'Section 1',
            content: 'Detailed content for section 1...'
        },
        {
            title: 'Section 2',
            content: 'More information here.'
        }
    ],
    headerColor: 0x2a2a3e,
    mode: 0
});

function animate() {
    requestAnimationFrame(animate);
    accordion.update();
    scene3D.render();
}
