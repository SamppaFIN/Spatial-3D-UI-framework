TextInput3D

Interactive logic-defying input fields with 3D presence

Live Demo

⚙️ Controls

Code Example

import { TextInput3D } from './src/controls/TextInput3D.js';

const inputField = new TextInput3D(scene, camera, [0, 0, 0], {
    value: 'Initial Text',
    placeholder: 'Type here...',
    width: 3.0,
    height: 0.8,
    type: 'text', // or 'password'
    mode: 0,
    style: 'post-it',
    onChange: (input, value) => {
        console.log('Value changed:', value);
    }
});

function animate() {
    requestAnimationFrame(animate);
    inputField.update();
    scene3D.render();
}
