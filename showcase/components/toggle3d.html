Toggle3D

Interactive 3D toggle switch with smooth animations and multi-state support

Live Demo

Status: OFF

⚙️ Controls

Code Example

import { Toggle3D } from './src/controls/Toggle3D.js';

const toggle = new Toggle3D(scene, camera, [0, 0, 0], {
    label: 'Notifications',
    onColor: 0x3366ff,
    offColor: 0x666666,
    isOn: true,
    trackShape: 'pill',
    animationPreset: 'bounce',
    width: 2.0,
    renderer: renderer,
    onClick: () => console.log('Toggled!')
});

function animate() {
    requestAnimationFrame(animate);
    toggle.update();
    scene3D.render();
}
